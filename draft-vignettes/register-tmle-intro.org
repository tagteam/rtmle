* About

The R-package =rtmle= implements a class of targeted minimum loss
estimators for the analysis of longitudinal register data. The
functionality is similar to the established R-package =ltmle= with the
following additional features and focus points:

- explicit handling of competing risks
- polypharmacy (more than one treatment variable are controlled by the regimes of the target trial)
- descriptive statistics
- diagnostic tools (not to be confused with model goodness of fit)
- subgroup analyses
- support for glmnet and ranger
- cheap bootstrap inference
- cross-fitting 
- parallel computing

* Data structures  

We consider register data in the following format.

    parameter_values = list(
        intercept_sex=qlogis(0.4),
        intercept_SBP=150,
        intercept_age=50,
        intercept_A=qlogis(0.3),
        intercept_B=qlogis(0.15),
        var_age=15,
        var_SBP=20,
        scale_death=0.01,
        scale_stroke=0.01,
        scale_bleeding=0.01,
        scale_dropout=0.01,
        effect_auto_A_A = 5,
        effect_auto_B_B = 4,
        effect_A_death = 0,
        effect_B_death = -0.8,
        effect_A_bleeding = 1.5,
        effect_B_bleeding = 0.5,
        effect_A_stroke = -0.5,
        effect_B_stroke = 0,
        effect_stroke_A = -2,
        effect_stroke_B = -1,

    )
)
ld <- register_format(cohort)

** Adding the data

#+ATTR_LATEX: :options otherkeywords={}, deletekeywords={}
#+BEGIN_SRC R  :results output verbatim drawer  :exports both  :session *R* :cache yes  
x <- add_long_data(x,
                   outcome_data = ld$timevar_data$stroke[!duplicated(id)], ## retain earliest stroke
                   censored_data = ld$timevar_data$dropout,
                   competing_data = ld$timevar_data$death,
                   timevar_data = ld$timevar_data[c("A", "B", "changeSBP", "bleeding")])
x <- add_baseline_data(x,data=ld$baseline_data)
x
#+END_SRC


** Preparing the data

#+ATTR_LATEX: :options otherkeywords={}, deletekeywords={}
#+BEGIN_SRC R  :results output verbatim drawer  :exports both  :session *R* :cache yes  
x <- discretize_data(x, start_followup_date = 0)
x <- prepare_rtmle_data(x)
x$prepared_data
#+END_SRC


** Defining the regimes

#+ATTR_LATEX: :options otherkeywords={}, deletekeywords={}
#+BEGIN_SRC R  :results output verbatim drawer  :exports both  :session *R* :cache yes  
x <- regime(x,name = "always A",treatment_variables = "A",intervention = 1)
x <- regime(x,name = "never A",treatment_variables = "A",intervention = 0)
x
#+END_SRC


** Defining the targets

#+ATTR_LATEX: :options otherkeywords={}, deletekeywords={}
#+BEGIN_SRC R  :results output verbatim drawer  :exports both  :session *R* :cache yes

x <- model_formula(x)
x
#+END_SRC
